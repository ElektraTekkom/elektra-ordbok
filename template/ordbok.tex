% Generated by build.py from the word files in ord/:
%   \ordbokentry{id}{ord}{(nynorsk) \textit{stikkord} $\bullet$ definisjon}
% Each entry leaves a mark where it starts ("entry number,letter number") and where it
% ends ("entry number"); the thumb index below uses them to find the letters on a page.
\newcount\ordbokentryno
\def\ordbokletterindex{0}
\newcommand{\ordbokentry}[3]{%
  \par\vspace{0.7em}%
  \hangindent=2em\hangafter=0%
  \global\advance\ordbokentryno 1\relax
  \markboth{#2}{#2}\hypertarget{#1}{}%
  \textbf{#2}%
  \begingroup\edef\x{\endgroup
    \noexpand\InsertMark{ordbokstart}{\the\ordbokentryno,\ordbokletterindex}}\x
  { }#3%
  \par
  \begingroup\edef\x{\endgroup\noexpand\InsertMark{ordbokend}{\the\ordbokentryno}}\x
}

% Thumb index (print version): green tabs on the outer edge of every word page, one
% for each letter that has text on the page (including an entry that runs over from
% the page before), lower down the page the further into the alphabet it is.
% entries.tex declares the letters (\ordboksettabs, \ordboktabletter) and says which
% letter the following entries belong to (\ordboksetletter).
\definecolor{elektragronn}{HTML}{009A44}
\NewMarkClass{ordbokstart}
\NewMarkClass{ordbokend}
\newif\ifordboktabs
\newcount\ordboktabi
\def\ordboktabcount{1}
\newcommand{\ordboksettabs}[1]{\gdef\ordboktabcount{#1}}
\newcommand{\ordboktabletter}[2]{\expandafter\gdef\csname ordboktab-#1\endcsname{#2}}
\newcommand{\ordboksetletter}[1]{\gdef\ordbokletterindex{#1}}
\def\ordbokparse#1,#2\relax{\def\ordbokn{#1}\def\ordbokk{#2}}
\newlength{\ordboktabh}
\newlength{\ordboktaby}
% The tab runs off the paper edge; the letter sits in the inner 8 mm, clear of the
% 3-5 mm most printers can't print.
\newcommand{\ordboktabw}{11mm}
\newcommand{\ordbokdrawtab}[1]{%
  \setlength{\ordboktabh}{\dimexpr(\paperheight-30mm)/\ordboktabcount\relax}%
  \setlength{\ordboktaby}{\dimexpr15mm+\ordboktabh*#1\relax}% bottom edge, from the top
  \def\ordboktabletterbox{\parbox[b][\ordboktabh][c]{8mm}{\centering
    \color{white}\sffamily\bfseries\large\csname ordboktab-#1\endcsname}}%
  \ifodd\value{page}% right-hand page: tab on the right edge, letter on its left part
    \put(\paperwidth-\ordboktabw,-\ordboktaby){%
      \textcolor{elektragronn}{\rule[0.4mm]{\ordboktabw}{\dimexpr\ordboktabh-0.8mm}}\llap{\ordboktabletterbox\hspace{3mm}}}%
  \else% left-hand page: tab on the left edge, letter on its right part
    \put(0pt,-\ordboktaby){%
      \textcolor{elektragronn}{\rule[0.4mm]{\ordboktabw}{\dimexpr\ordboktabh-0.8mm}}\llap{\ordboktabletterbox}}%
  \fi}
\ifdefined\ordbokprint
\AddToHook{shipout/background}{%
  \ifordboktabs
    \def\ordboklo{0}\def\ordbokhi{0}%
    % An entry that started before this page and hasn't ended yet runs onto it.
    \edef\ordboktmp{\TopMark{ordbokstart}}%
    \ifx\ordboktmp\empty\else
      \expandafter\ordbokparse\ordboktmp\relax
      \edef\ordbokendn{\TopMark{ordbokend}}%
      \ifnum0\ordbokendn<\ordbokn\relax
        \edef\ordboklo{\ordbokk}\edef\ordbokhi{\ordbokk}%
      \fi
    \fi
    % Entries that start on this page: from the first one's letter to the last one's.
    \IfMarksEqualTF{ordbokstart}{top}{first}{}{%
      \edef\ordboktmp{\FirstMark{ordbokstart}}\expandafter\ordbokparse\ordboktmp\relax
      \ifnum\ordboklo=0 \edef\ordboklo{\ordbokk}\fi
      \edef\ordboktmp{\LastMark{ordbokstart}}\expandafter\ordbokparse\ordboktmp\relax
      \edef\ordbokhi{\ordbokk}}%
    \ifnum\ordboklo>0
      \ordboktabi=\ordboklo\relax
      \loop
        \ordbokdrawtab{\the\ordboktabi}%
      \ifnum\ordboktabi<\ordbokhi\relax
        \advance\ordboktabi by 1
      \repeat
    \fi
  \fi}
\fi
